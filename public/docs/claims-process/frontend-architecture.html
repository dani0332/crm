⚡ Vue 3 Composition API Patterns

🎯 Modern Component Structure

All components use the Composition API with <script setup> syntax for better TypeScript support and cleaner code organization.

// ClaimDetails.vue - Composition API Example <script setup> import { ref, computed, watch, onMounted } from 'vue'; import { usePage, useForm, router } from '@inertiajs/vue3'; import { useCan, useCanAny } from '@/composables/permissions'; // Props definition with TypeScript-like validation const props = defineProps({ claim: { type: Object, required: true, }, documentTypes: { type: Array, default: () => [], }, }); // Emits definition for parent communication const emit = defineEmits(['updated', 'statusChanged']); // Reactive state management const isEditing = ref(false); const loading = ref(false); const errors = ref({}); // Permission checks using composables const canEdit = useCan(permissionsEnum.CLAIM_EDIT); const canUpdateStatus = useCan(permissionsEnum.CLAIMS_STATUS_UPDATE); // Computed properties for reactive data const claimStatus = computed(() => { return props.claim.claim_status?.name || 'Unknown'; }); const isCarClaim = computed(() => { return props.claim.quote_type_id === QuoteTypes.CAR.id(); }); // Form handling with Inertia const form = useForm({ incident_date: props.claim.incident_date, incident_story: props.claim.incident_story, plate_number: props.claim.claim_request_detail?.plate_number || '', car_make: props.claim.claim_request_detail?.car_make || '', car_model: props.claim.claim_request_detail?.car_model || '', model_year: props.claim.claim_request_detail?.model_year || '', }); // Watchers for reactive updates watch(() => props.claim, (newClaim) => { // Update form when claim prop changes form.defaults(extractFormData(newClaim)); form.reset(); }, { deep: true }); // Methods const submitForm = () => { if (!canEdit.value) return; loading.value = true; form.post(route('claims.update.details', props.claim.uuid), { onSuccess: () => { isEditing.value = false; emit('updated', props.claim); toast.success('Claim details updated successfully'); }, onError: (errors) => { console.error('Update failed:', errors); toast.error('Failed to update claim details'); }, onFinish: () => { loading.value = false; }, }); }; // Lifecycle hooks onMounted(() => { // Initialize component console.log('ClaimDetails component mounted'); }); </script> <template> <div class="claim-details-container"> <!-- Permission-based rendering --> <div v-if="canEdit" class="edit-controls"> <button @click="isEditing = !isEditing" class="btn btn-primary"> {{ isEditing ? 'Cancel' : 'Edit Details' }} </button> </div> <!-- Conditional form rendering --> <form v-if="isEditing" @submit.prevent="submitForm" class="claim-form"> <!-- LOB-specific fields --> <div v-if="isCarClaim" class="car-fields-section"> <input v-model="form.plate_number" placeholder="Plate Number" /> <input v-model="form.car_make" placeholder="Car Make" /> <input v-model="form.car_model" placeholder="Car Model" /> <input v-model="form.model_year" type="number" placeholder="Model Year" /> </div> <button type="submit" :disabled="form.processing || loading"> {{ loading ? 'Updating...' : 'Update Details' }} </button> </form> <!-- Display mode --> <div v-else class="claim-display"> <!-- Claim information display --> </div> </div> </template>

🔐 Permission-Based Rendering

🛡️ Security-First UI Components

All components implement granular permission checks to ensure users only see and can interact with features they're authorized to use.

🔑 Permission Composables

  • useCan() - Single permission check
  • useCanAny() - Multiple permission OR logic
  • useCanAll() - Multiple permission AND logic
  • Reactive permission updates

🎭 Conditional Rendering

  • v-if for permission-based visibility
  • Disabled states for unauthorized actions
  • Dynamic menu items based on permissions
  • Role-based component variations
// composables/permissions.js - Permission management import { computed } from 'vue'; import { usePage } from '@inertiajs/vue3'; export function useCan(permission) { const page = usePage(); return computed(() => { const userPermissions = page.props.auth?.permissions || []; return userPermissions.includes(permission); }); } export function useCanAny(permissions) { const page = usePage(); return computed(() => { const userPermissions = page.props.auth?.permissions || []; return permissions.some(permission => userPermissions.includes(permission)); }); } // Usage in components const canViewClaims = useCan(permissionsEnum.CLAIM_LIST); const canManageClaims = useCanAny([ permissionsEnum.CLAIM_CREATE, permissionsEnum.CLAIM_EDIT, permissionsEnum.CLAIM_DELETE ]);
