🔍 Internal Policy Search Integration

📋 System Database Queries

Policy search is handled internally within the current system using database queries on the PersonalQuote model, providing fast and reliable policy validation for claim creation.

🎯 Internal Search Implementation

POST /claim/search-policies

PersonalQuote::where('status', 'PolicyBooked')

Optimized queries with relationship loading
// ClaimsService::searchActivePolicies() - Internal Database Integration public function searchActivePolicies( ?string $email = null, ?string $policyNumber = null, ?int $quoteTypeId = null, int $page = 1 ): array { try { // Build internal database query $query = PersonalQuote::with(['customer', 'quoteType']) ->where('status', 'PolicyBooked') ->select([ 'id', 'uuid', 'ref_id', 'policy_number', 'customer_id', 'quote_type_id', 'car_make', 'car_model', 'model_year', 'plate_number', 'status', 'created_at' ]); // Apply search filters if ($email) { $query->whereHas('customer', function($q) use ($email) { $q->where('email', 'like', "%{$email}%"); }); } if ($policyNumber) { $query->where('policy_number', 'like', "%{$policyNumber}%"); } if ($quoteTypeId) { $query->where('quote_type_id', $quoteTypeId); } // Execute paginated query $policies = $query->simplePaginate($this->perPage, ['*'], 'page', $page); LoggerService::info('Internal policy search successful', [ 'search_params' => compact('email', 'policyNumber', 'quoteTypeId', 'page'), 'result_count' => $policies->count(), 'user_id' => Auth::id(), ]); return [ 'policies' => $policies->items(), 'pagination' => [ 'current_page' => $policies->currentPage(), 'per_page' => $policies->perPage(), 'total' => $policies->total(), 'has_more' => $policies->hasMorePages(), ] ]; } catch (\Exception $e) { LoggerService::error(self::class.'::'.__FUNCTION__.' - Internal search error', [ 'error' => $e->getMessage(), 'search_params' => compact('email', 'policyNumber', 'quoteTypeId', 'page'), 'user_id' => Auth::id(), ]); throw $e; } }

🔐 Internal System Security

  • Database Security: Eloquent ORM with parameterized queries
  • Access Control: Permission-based policy search access
  • Performance: Optimized queries with eager loading
  • Audit Logging: All search activities logged for compliance

🔌 CAPI (Customer Portal API) Integration

📋 Claim Creation Functionality

CAPI integration is used exclusively for claim creation, allowing the system to create claims in the external system with comprehensive data mapping and error handling.

🎯 CAPI Endpoints Used

POST /api/claims/save-claim
Create new claim in external system
// ClaimsService::createClaim() - CAPI Integration for Claim Creation public function createClaim(array $claimData): array { try { // Prepare CAPI data with 15+ mapped fields $capiData = [ 'first_name' => $claimData['first_name'], 'last_name' => $claimData['last_name'], 'email' => $claimData['email'], 'mobile_no' => $claimData['mobile_no'], 'quote_type_id' => $claimData['quote_type_id'], 'claim_type_id' => $claimData['claim_type_id'], 'incident_date' => $claimData['incident_date'], 'incident_story' => $claimData['incident_story'], 'policy_number' => $claimData['policy_number'], 'customer_id' => $claimData['customer_id'], 'insurance_provider_id' => $claimData['insurance_provider_id'], 'selected_quote_uuid' => $claimData['selected_quote_uuid'], 'car_make' => $claimData['car_make'] ?? null, 'car_model' => $claimData['car_model'] ?? null, 'model_year' => $claimData['model_year'] ?? null, 'plate_number' => $claimData['plate_number'] ?? null, 'claim_request_type_id' => $claimData['claim_request_type_id'], 'service_type_id' => $claimData['service_type_id'], 'request_reference_number' => $claimData['request_reference_number'], ]; // Make CAPI request for claim creation $response = CustomerPortalApiFacade::request('/api/claims/save-claim', 'post', $capiData); if ($response->success) { LoggerService::info('CAPI claim creation successful', [ 'claim_data' => $capiData, 'response' => $response->data, 'user_id' => Auth::id(), ]); return [ 'success' => true, 'data' => $response->data, 'message' => 'Claim created successfully via CAPI' ]; } throw new \Exception('CAPI claim creation failed: ' . $response->message); } catch (\Exception $e) { LoggerService::error(self::class.'::'.__FUNCTION__.' - CAPI claim creation error', [ 'error' => $e->getMessage(), 'claim_data' => $claimData ?? [], 'user_id' => Auth::id(), ]); throw $e; } }

🔐 CAPI Authentication

  • Authentication Type: Bearer Token + API Key
  • Token Refresh: Automatic token renewal on expiry
  • Rate Limiting: 100 requests per minute per user
  • Timeout Settings: 30 seconds connection, 60 seconds read

// ClaimsService::optimizeMessageWithAI() - AI Integration public function optimizeMessageWithAI(string $message, string $claimUuid): array { try { $apiData = [ 'original_message' => $message, 'claim_reference' => $claimUuid, 'optimization_level' => 'professional', 'target_audience' => 'insurance_customer', 'preserve_meaning' => true, ]; // Make AI optimization request $response = InstantWriterAIFacade::request('/message-optimizer/optimize', 'post', $apiData); if ($response->success && isset($response->data['optimized_message'])) { LoggerService::info('AI message optimization successful', [ 'claim_uuid' => $claimUuid, 'original_length' => strlen($message), 'optimized_length' => strlen($response->data['optimized_message']), 'user_id' => Auth::id(), ]); return [ 'success' => true, 'original_message' => $message, 'optimized_message' => $response->data['optimized_message'], 'improvements' => $response->data['improvements'] ?? [], 'confidence_score' => $response->data['confidence_score'] ?? null, ]; } throw new \Exception('AI optimization failed: ' . ($response->message ?? 'Unknown error')); } catch (\Exception $e) { LoggerService::error(self::class.'::'.__FUNCTION__.' - AI optimization error', [ 'error' => $e->getMessage(), 'claim_uuid' => $claimUuid, 'message_length' => strlen($message), 'user_id' => Auth::id(), ]); // Return original message if AI fails return [ 'success' => false, 'original_message' => $message, 'optimized_message' => $message, 'error' => $e->getMessage(), ]; } }
