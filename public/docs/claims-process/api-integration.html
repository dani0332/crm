🔌 CAPI (Customer Portal API) Integration

📋 Policy Search Functionality

CAPI integration provides policy search capabilities for claim creation, allowing claims managers to find and validate active policies before creating claims.

🎯 CAPI Endpoints Used

POST /api/claims/save-claim
Create new claim in external system
GET /api/policies/search
Search active policies by email/policy number
GET /api/customers/verify
Verify customer information
// ClaimsService::searchActivePolicies() - CAPI Integration public function searchActivePolicies( ?string $email = null, ?string $policyNumber = null, ?int $quoteTypeId = null, int $page = 1 ): array { try { // Prepare search parameters $searchParams = array_filter([ 'email' => $email, 'policy_number' => $policyNumber, 'quote_type_id' => $quoteTypeId, 'page' => $page, 'per_page' => $this->perPage, ]); // Make CAPI request $response = CustomerPortalApiFacade::request('/api/policies/search', 'get', $searchParams); if ($response->success) { LoggerService::info('CAPI policy search successful', [ 'search_params' => $searchParams, 'result_count' => count($response->data), 'user_id' => Auth::id(), ]); return $response->data; } throw new \Exception('CAPI policy search failed: ' . $response->message); } catch (\Exception $e) { LoggerService::error(self::class.'::'.__FUNCTION__.' - CAPI search error', [ 'error' => $e->getMessage(), 'search_params' => $searchParams ?? [], 'user_id' => Auth::id(), ]); throw $e; } }

🔐 CAPI Authentication

  • Authentication Type: Bearer Token + API Key
  • Token Refresh: Automatic token renewal on expiry
  • Rate Limiting: 100 requests per minute per user
  • Timeout Settings: 30 seconds connection, 60 seconds read

// ClaimsService::optimizeMessageWithAI() - AI Integration public function optimizeMessageWithAI(string $message, string $claimUuid): array { try { $apiData = [ 'original_message' => $message, 'claim_reference' => $claimUuid, 'optimization_level' => 'professional', 'target_audience' => 'insurance_customer', 'preserve_meaning' => true, ]; // Make AI optimization request $response = InstantWriterAIFacade::request('/message-optimizer/optimize', 'post', $apiData); if ($response->success && isset($response->data['optimized_message'])) { LoggerService::info('AI message optimization successful', [ 'claim_uuid' => $claimUuid, 'original_length' => strlen($message), 'optimized_length' => strlen($response->data['optimized_message']), 'user_id' => Auth::id(), ]); return [ 'success' => true, 'original_message' => $message, 'optimized_message' => $response->data['optimized_message'], 'improvements' => $response->data['improvements'] ?? [], 'confidence_score' => $response->data['confidence_score'] ?? null, ]; } throw new \Exception('AI optimization failed: ' . ($response->message ?? 'Unknown error')); } catch (\Exception $e) { LoggerService::error(self::class.'::'.__FUNCTION__.' - AI optimization error', [ 'error' => $e->getMessage(), 'claim_uuid' => $claimUuid, 'message_length' => strlen($message), 'user_id' => Auth::id(), ]); // Return original message if AI fails return [ 'success' => false, 'original_message' => $message, 'optimized_message' => $message, 'error' => $e->getMessage(), ]; } }

// Comprehensive error handling with retry logic public function makeApiRequestWithRetry(string $endpoint, string $method, array $data, int $maxRetries = 3): array { $attempt = 0; $lastException = null; while ($attempt < $maxRetries) { try { $response = $this->makeApiRequest($endpoint, $method, $data); // Log successful request after retries if ($attempt > 0) { LoggerService::info('API request succeeded after retry', [ 'endpoint' => $endpoint, 'attempt' => $attempt + 1, 'user_id' => Auth::id(), ]); } return $response; } catch (\Exception $e) { $attempt++; $lastException = $e; // Determine if error is retryable if (!$this->isRetryableError($e)) { break; } // Exponential backoff $delay = pow(2, $attempt - 1); sleep($delay); LoggerService::warning('API request failed, retrying', [ 'endpoint' => $endpoint, 'attempt' => $attempt, 'error' => $e->getMessage(), 'retry_delay' => $delay, ]); } } // All retries exhausted LoggerService::error('API request failed after all retries', [ 'endpoint' => $endpoint, 'max_retries' => $maxRetries, 'final_error' => $lastException->getMessage(), 'user_id' => Auth::id(), ]); throw $lastException; } private function isRetryableError(\Exception $e): bool { // Network errors, timeouts, and 5xx server errors are retryable return in_array($e->getCode(), [500, 502, 503, 504, 0]) || strpos($e->getMessage(), 'timeout') !== false || strpos($e->getMessage(), 'connection') !== false; }

// Caching implementation for policy search public function searchActivePoliciesCached(array $searchParams): array { $cacheKey = 'policy_search_' . md5(serialize($searchParams)); return Cache::remember($cacheKey, now()->addMinutes(15), function () use ($searchParams) { return $this->searchActivePolicies( $searchParams['email'] ?? null, $searchParams['policy_number'] ?? null, $searchParams['quote_type_id'] ?? null, $searchParams['page'] ?? 1 ); }); } // Rate limiting implementation public function checkRateLimit(string $userId, string $endpoint): bool { $key = "api_rate_limit_{$userId}_{$endpoint}"; $attempts = Cache::get($key, 0); if ($attempts >= config('api.rate_limit.max_attempts', 60)) { LoggerService::warning('API rate limit exceeded', [ 'user_id' => $userId, 'endpoint' => $endpoint, 'attempts' => $attempts, ]); return false; } Cache::put($key, $attempts + 1, now()->addMinute()); return true; }

// API monitoring and metrics collection public function logApiMetrics(string $endpoint, float $responseTime, bool $success, ?string $error = null): void { $metrics = [ 'endpoint' => $endpoint, 'response_time_ms' => round($responseTime * 1000, 2), 'success' => $success, 'timestamp' => now()->toISOString(), 'user_id' => Auth::id(), ]; if (!$success && $error) { $metrics['error'] = $error; } // Log to metrics collection system LoggerService::info('API request metrics', $metrics); // Store in performance database for analytics ApiMetric::create($metrics); // Check for performance degradation $this->checkPerformanceAlerts($endpoint, $responseTime, $success); } private function checkPerformanceAlerts(string $endpoint, float $responseTime, bool $success): void { // Alert if response time exceeds threshold if ($responseTime > config('api.performance.response_time_threshold', 5.0)) { event(new ApiPerformanceAlert($endpoint, $responseTime, 'slow_response')); } // Alert if error rate is high $recentErrorRate = $this->calculateRecentErrorRate($endpoint); if ($recentErrorRate > config('api.performance.error_rate_threshold', 0.1)) { event(new ApiPerformanceAlert($endpoint, $recentErrorRate, 'high_error_rate')); } }
