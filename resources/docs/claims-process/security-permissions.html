🎯 Granular Permission System

📋 Claims-Specific Permissions

The system implements 11 specific permissions for claims management, providing fine-grained access control for different user roles.

Permission Scope Description Typical Roles
CLAIM_LIST Read View claims listing and search functionality All Claims Staff
CLAIM_SHOW Read View individual claim details and information Claims Handlers, Managers
CLAIM_CREATE Write Create new claims via CPAPI integration Claims Handlers
CLAIM_EDIT Write Edit existing claim details and information Claims Handlers, Senior Staff
CLAIMS_STATUS_UPDATE Write Update main claim status (Open/Closed) Team Leads, Managers
CLAIMS_SUB_STATUS_UPDATE Write Update detailed sub-status and send notifications Claims Managers
CLAIM_DOCUMENT_UPLOAD Write Upload documents to claims Claims Handlers
CLAIM_DOCUMENT_DELETE Delete Remove documents from claims Senior Claims Staff
CLAIM_DOCUMENT_S3_URL Access Generate secure S3 temporary URLs Claims Staff
CLAIM_DOWNLOAD_ALL_DOCUMENTS Bulk Download all claim documents as ZIP Claims Managers
CLAIMS_EXPORT_DATA Export Export claims data for reporting Managers, Analysts

🔒 Implementation Patterns

⚙️ Backend Permission Enforcement

// Controller-level permission middleware class ClaimsController extends Controller { public function __construct() { // Apply permission middleware to all methods $this->middleware('permission:'.PermissionsEnum::CLAIM_LIST)->only(['index']); $this->middleware('permission:'.PermissionsEnum::CLAIM_SHOW)->only(['show']); $this->middleware('permission:'.PermissionsEnum::CLAIM_CREATE)->only(['create', 'store']); $this->middleware('permission:'.PermissionsEnum::CLAIM_EDIT)->only(['edit', 'update']); $this->middleware('permission:'.PermissionsEnum::CLAIMS_STATUS_UPDATE)->only(['updateClaimStatus']); $this->middleware('permission:'.PermissionsEnum::CLAIMS_SUB_STATUS_UPDATE)->only(['sendNotification']); } // Method-level permission checks public function updateClaimDetails(ClaimDetailsUpdateRequest $request, ClaimRequest $claim): RedirectResponse { // Additional permission validation if needed if (!Auth::user()->can(PermissionsEnum::CLAIM_EDIT)) { abort(403, 'Unauthorized action.'); } // Log permission usage for audit LoggerService::info('Permission used: CLAIM_EDIT', [ 'user_id' => Auth::id(), 'claim_uuid' => $claim->uuid, 'action' => 'update_details', ]); // Proceed with update logic return $this->claimsService->updateClaimDetails($claim, $request->validated()); } }

🎨 Frontend Permission Integration

// Vue.js permission-based rendering <script setup> import { useCan, useCanAny } from '@/composables/permissions'; import { PermissionsEnum } from '@/enums/PermissionsEnum'; // Permission checks const canViewClaims = useCan(PermissionsEnum.CLAIM_LIST); const canCreateClaims = useCan(PermissionsEnum.CLAIM_CREATE); const canEditClaims = useCan(PermissionsEnum.CLAIM_EDIT); const canManageStatus = useCanAny([ PermissionsEnum.CLAIMS_STATUS_UPDATE, PermissionsEnum.CLAIMS_SUB_STATUS_UPDATE ]); const canManageDocuments = useCanAny([ PermissionsEnum.CLAIM_DOCUMENT_UPLOAD, PermissionsEnum.CLAIM_DOCUMENT_DELETE ]); </script> <template> <div class="claims-interface"> <!-- Navigation based on permissions --> <nav class="claims-nav"> <router-link v-if="canViewClaims" to="/claims" class="nav-link"> View Claims </router-link> <router-link v-if="canCreateClaims" to="/claims/create" class="nav-link"> Create Claim </router-link> </nav> <!-- Conditional feature rendering --> <div class="claim-actions"> <button v-if="canEditClaims" @click="editClaim" class="btn btn-primary"> Edit Details </button> <ClaimStatus v-if="canManageStatus" :claim="claim" /> <ClaimDocuments v-if="canManageDocuments" :claim="claim" /> <!-- Disabled state for unauthorized users --> <button v-else disabled class="btn btn-disabled" title="Insufficient permissions"> Edit Details </button> </div> </div> </template>

'system', 'ip_address' => request()->ip(), 'user_agent' => request()->userAgent(), 'changes' => $changes, 'timestamp' => now()->toISOString(), 'action' => 'claim_update', 'session_id' => session()->getId(), ]); // Create activity record for user interface ClaimActivity::createForClaim( $claimRequest->id, $claimRequest->uuid, $claimRequest->claim_status_id, 'Claim details updated', null, Auth::id() ); } public function created(ClaimRequest $claimRequest): void { LoggerService::info('New claim created', [ 'claim_uuid' => $claimRequest->uuid, 'user_id' => Auth::id(), 'customer_email' => $claimRequest->email, 'quote_type_id' => $claimRequest->quote_type_id, 'source' => $claimRequest->source, 'ip_address' => request()->ip(), 'timestamp' => now()->toISOString(), 'action' => 'claim_creation', ]); } }

📁 Secure File Handling

// Secure document access with temporary URLs class ClaimsService extends BaseService { public function generateSecureDocumentUrl(int $documentId): array { $document = GenericDocument::findOrFail($documentId); // Verify user has permission to access this document $this->authorize('view', $document); // Generate temporary URL with expiration $temporaryUrl = Storage::disk('s3')->temporaryUrl( $document->file_path, now()->addHour(), [ 'ResponseContentType' => $document->mime_type, 'ResponseContentDisposition' => 'inline; filename="'.$document->file_name.'"', ] ); // Log document access for audit LoggerService::info('Document accessed', [ 'document_id' => $documentId, 'document_name' => $document->file_name, 'claim_uuid' => $document->model->uuid ?? null, 'user_id' => Auth::id(), 'ip_address' => request()->ip(), 'access_method' => 'temporary_url', 'expires_at' => now()->addHour()->toISOString(), ]); return [ 'url' => $temporaryUrl, 'expires_at' => now()->addHour(), 'document_name' => $document->file_name, ]; } public function validateFileUpload(UploadedFile $file): bool { // File type validation $allowedMimes = ['pdf', 'jpg', 'jpeg', 'png', 'doc', 'docx']; if (!in_array($file->getClientOriginalExtension(), $allowedMimes)) { throw new ValidationException('Invalid file type'); } // File size validation (10MB limit) if ($file->getSize() > 10 * 1024 * 1024) { throw new ValidationException('File size exceeds limit'); } // Virus scanning (if enabled) if (config('security.virus_scanning_enabled')) { $scanResult = $this->performVirusScan($file); if (!$scanResult['clean']) { LoggerService::error('Virus detected in uploaded file', [ 'filename' => $file->getClientOriginalName(), 'user_id' => Auth::id(), 'scan_result' => $scanResult, ]); throw new SecurityException('File failed security scan'); } } return true; } }
